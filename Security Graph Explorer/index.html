Config JSON — Schema Reference
// Full schema
{
  "title": "string",

  "nodes": [  // one entry per entity column
    {
      "id_column": "col_name",    // required — unique node ID source
      "label_column": "col_name", // optional — display label (default: id_column)
      "type": "user|machine|ip|alert", // fixed type string
      "type_column": "col_name"   // OR read type dynamically from row
    }
  ],

  "edges": [  // one entry per relationship type
    {
      "from_column": "col_name",  // required
      "to_column": "col_name",    // required
      "label_column": "col_name",  // optional — value shown on edge
      "label": "static label",     // optional — fixed label for all edges
      "color": "#rrggbb",         // optional — fixed edge color
      "color_map": {              // optional — map label values to colors
        "value1": "#rrggbb",
        "value2": "#rrggbb"
      }
    }
  ]
}

// Built-in node types → style
user    → blue ellipse
machine → green box
ip      → yellow diamond
alert   → red triangle
(other) → purple dot
